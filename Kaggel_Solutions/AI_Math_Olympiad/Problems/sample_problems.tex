\section*{Problem 1: The Circle Intersection}

Let $C_1$ be the circle defined by $(x-2)^2 + (y-3)^2 = 25$ and $C_2$ be the circle defined by $(x-8)^2 + (y-11)^2 = 100$. 
Determine the number of intersection points between $C_1$ and $C_2$.

\section*{Problem 2: The Integer Sequence}
Let $a_n$ be a sequence such that $a_1 = 1$ and for $n \ge 1$, 
\[ a_{n+1} = \frac{a_n}{1 + n a_n}. \]
Find $a_{2024}$.
